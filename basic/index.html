    #include <LiquidCrystal_I2C.h>
LiquidCrystal_I2C lcd(0x27,16,2);

// LCD刷新定时
unsigned long lastLcdTime = 0;
const unsigned long lcdInterval = 200;  // 每200ms刷新一次，兼顾流畅度和无闪烁

// 引脚定义
#define TRIG 9
#define ECHO 8
#define LED_GREEN 7
#define LED_YELLOW 6
#define LED_RED 5
#define BUZZER 4
#define KEY 2

// 全局变量
float d = 0;          // 测距原始值
int dis = 0;          // 取整后的距离，给屏幕显示
bool isAlarm = true;  // true=ALARM报警模式，false=MUTE静音，上电默认true
String statusStr;     // 状态字符串 SAFE/CAUTION/WARNING/STOP

// 按键防抖变量
int lastKeyState = HIGH;
unsigned long lastDebounceTime = 0;
const unsigned long debounceDelay = 20;

// 超声波定时
unsigned long lastUltraTime = 0;
const unsigned long ultraInterval = 200;

// 红灯闪烁计时变量
unsigned long lastRedTime = 0;
unsigned long redInterval = 250;
bool redOn = false;

// 蜂鸣器计时变量
unsigned long lastBuzzTime = 0;
unsigned long buzzInterval = 500;
bool buzzOn = false;

void setup() {
  Serial.begin(9600);
  lcd.init();
  lcd.backlight();
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
  pinMode(LED_GREEN, OUTPUT);
  pinMode(LED_YELLOW, OUTPUT);
  pinMode(LED_RED, OUTPUT);
  pinMode(BUZZER, OUTPUT);
  pinMode(KEY, INPUT_PULLUP); //内部上拉
}

// 超声波测距函数，返回厘米
float getDistance(){
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  long time = pulseIn(ECHO, HIGH);
  float dist = time * 0.034 / 2;
  return dist;
}

void loop() {
  // 1. 定时读取超声波距离（200ms一次，避免高频读取出错）
  if(millis() - lastUltraTime > ultraInterval){
    lastUltraTime = millis();
    d = getDistance();
    // 异常值限幅，防止数值错乱
    if(d < 0) d = 0;
    if(d > 200) d = 200;
    dis = int(d); // 转整数，满足题目显示要求
  }
  // 2. 按键防抖 + 静音切换（仅WARNING/STOP档位有效）
  keyScan();
  // 3. 状态判断，更新状态文本、闪烁周期
  stateJudge();
  // 4. 灯光控制（常亮/固定闪烁）
  ledControl();
  // 5. 蜂鸣器控制（ALARM模式生效，MUTE模式静音）
  buzzerControl();
  // 6. 定时刷新LCD屏幕，解决闪烁问题
  if(millis() - lastLcdTime > lcdInterval){
    lastLcdTime = millis();
    lcdShow();
  }
}

// ========== 按键扫描防抖函数 ==========
void keyScan(){
  int currentKey = digitalRead(KEY);
  if(currentKey != lastKeyState){
    lastDebounceTime = millis();
  }
  if(millis() - lastDebounceTime > debounceDelay){
    if(currentKey != lastKeyState){
      lastKeyState = currentKey;
      // 按下是LOW，并且当前状态是WARNING或者STOP，才切换模式
      if(currentKey == LOW){
        if( (dis >=20 && dis <50) || (dis <20) ){
          isAlarm = !isAlarm;
        }
      }
    }
  }
}

// ========== 状态判断函数 ==========
void stateJudge(){
  if(dis > 100){
    statusStr = "SAFE";
    redInterval = 250;
    buzzInterval = 500;
  }else if(dis >=50 && dis <=100){
    statusStr = "CAUTION";
    redInterval = 250;
    buzzInterval = 500;
  }else if(dis >=20 && dis <50){
    statusStr = "WARNING";
    redInterval = 250;
    buzzInterval = 500;
  }else if(dis < 20){
    statusStr = "STOP";
    redInterval = 125;
    buzzInterval = 125;
  }
}

// ========== LED灯光控制 ==========
void ledControl(){
  // 先全部熄灭
  digitalWrite(LED_GREEN, LOW);
  digitalWrite(LED_YELLOW, LOW);
  digitalWrite(LED_RED, LOW);
  if(statusStr == "SAFE"){
    digitalWrite(LED_GREEN, HIGH);
  }else if(statusStr == "CAUTION"){
    digitalWrite(LED_YELLOW, HIGH);
  }else if(statusStr == "WARNING"){
    if(millis() - lastRedTime > redInterval){
      lastRedTime = millis();
      redOn = !redOn;
      digitalWrite(LED_RED, redOn);
    }
  }else if(statusStr == "STOP"){
    if(millis() - lastRedTime > redInterval){
      lastRedTime = millis();
      redOn = !redOn;
      digitalWrite(LED_RED, redOn);
    }
  }
}

// ========== 蜂鸣器控制 ==========
void buzzerControl(){
  //静音模式，直接关闭蜂鸣器
  if(isAlarm == false){
    digitalWrite(BUZZER, LOW);
    buzzOn = false;
    return;
  }
  //ALARM模式，只有WARNING STOP才响
  if(statusStr == "WARNING" || statusStr == "STOP"){
    if(millis() - lastBuzzTime > buzzInterval){
      lastBuzzTime = millis();
      buzzOn = !buzzOn;
      digitalWrite(BUZZER, buzzOn);
    }
  }else{
    digitalWrite(BUZZER, LOW);
    buzzOn = false;
  }
}

// ========== LCD显示 ==========
void lcdShow(){
  lcd.setCursor(0, 0);
  lcd.print(dis);
  lcd.print(" cm ");
  lcd.print(statusStr);
  lcd.print(" ");
  if(isAlarm){
    lcd.print("ALARM  ");
  }else{
    lcd.print("MUTE   ");
  }
}
